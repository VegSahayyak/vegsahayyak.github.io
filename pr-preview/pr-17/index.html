dependencies {
    // 1. Bill of Materials for version alignment
    implementation(platform("com.vegsahayyak:speed-bom:2.1.0"))
    
    // 2. Headless tracking & alarm logic
    implementation("com.vegsahayyak:speed-core")
    
    // 3. Optional Jetpack Compose Speedometer & HUD UI
    implementation("com.vegsahayyak:speed-ui")

    // 4. Optional PiP floating overlay & Activity Recognition plugins
    implementation("com.vegsahayyak:speed-overlay")
    implementation("com.vegsahayyak:speed-autostart")
}
kotlin
class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        // 1. Configure and launch headless tracking
        val assistant = SpeedAssistant.with(this)
            .setWarningSpeedThreshold(80f) // Alert threshold in km/h
            .setMinWarningSpeedThreshold(40f) // Minimum corridor threshold

        assistant.start()

        // 2. Render Jetpack Compose UI
        setContent {
            val speedState by SpeedAssistant.observeFlow().collectAsState(initial = SpeedData.SpeedUnavailable)
            val currentSpeedKmh = (speedState as? SpeedData.SpeedChanged)?.let { it.speedMps * 3.6f } ?: 0f
            val isOverLimit = (speedState as? SpeedData.SpeedChanged)?.isOverLimit ?: false
            
            SpeedGauge(
                speed = currentSpeedKmh,
                warningSpeed = 80f,
                isTrackingActive = true,
                isOverLimit = isOverLimit,
                unitLabel = "km/h"
            )
        }
    }
}
